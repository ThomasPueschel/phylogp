# Stan code for a kernel

Writes the complete Stan program that
[`phylo_gp()`](https://thomaspueschel.github.io/phylogp/reference/phylo_gp.md)
compiles for a kernel. The program can be inspected, edited and run
directly with `cmdstanr` together with the data from
[`make_standata()`](https://thomaspueschel.github.io/phylogp/reference/make_standata.md).

## Usage

``` r
make_stancode(kernel, tree = NULL)
```

## Arguments

- kernel:

  A kernel, e.g. `k_bm() + k_white()`.

- tree:

  Optional tree. It decides whether components such as
  [`k_ou()`](https://thomaspueschel.github.io/phylogp/reference/classical-kernels.md)
  can use the branch-wise form (ultrametric trees) or need dense
  algebra. Without a tree an ultrametric tree is assumed.

## Value

A character string with class `phylogp_stancode`.

## Examples

``` r
cat(make_stancode(k_bm() + k_white()))
#> // Generated by the phylogp R package (phylogenetic Gaussian processes)
#> // Kernel: bm() + white()
#> // Likelihood: pruning (O(N))
#> functions {
#>   // ---------------------------------------------------------------------------
#>   // phylogp: shared Stan functions.
#>   // Tree kernels of branch-wise form
#>   //   K = B diag(w) B' + rv 11' + diag(t2)
#>   // (w: edge variances, rv: root variance, t2: extra tip variance).
#>   // Edges are in ape "postorder" (children before parents), so a single pass
#>   // over the edges gives the O(N) pruning likelihood.
#>   // ---------------------------------------------------------------------------
#> 
#>   // Dense kernel matrix  K = B diag(w) B' + rv + diag(t2)
#>   matrix tree_kernel(matrix B, vector w, real rv, vector t2) {
#>     matrix[rows(B), rows(B)] K = diag_post_multiply(B, w) * B';
#>     return add_diag(K + rv, t2);
#>   }
#> 
#>   // O(N) Gaussian pruning log density of residuals r
#>   // (Felsenstein 1973; Ho & Ane 2014)
#>   real tree_pruning_lpdf(vector r, vector w, real rv, vector t2,
#>                          array[] int anc, array[] int des,
#>                          array[] int first_child, int n_nodes, int root) {
#>     int N = num_elements(r);
#>     int E = num_elements(w);
#>     vector[n_nodes] m;
#>     vector[n_nodes] v;
#>     real lp = 0;
#>     m[1:N] = r;
#>     v[1:N] = t2;
#>     for (e in 1:E) {
#>       int c = des[e];
#>       int p = anc[e];
#>       real vc = v[c] + w[e];
#>       if (first_child[e] == 1) {
#>         m[p] = m[c];
#>         v[p] = vc;
#>       } else {
#>         real s = v[p] + vc;
#>         lp += normal_lpdf(m[p] | m[c], sqrt(s));
#>         m[p] = (m[p] * vc + m[c] * v[p]) / s;
#>         v[p] = v[p] * vc / s;
#>       }
#>     }
#>     lp += normal_lpdf(0 | m[root], sqrt(v[root] + rv));
#>     return lp;
#>   }
#> 
#>   // Moments of the latent tip value x_i given y_{-i}, for every tip, by an
#>   // upward-downward pass of Gaussian messages (O(N)). Column 1: mean,
#>   // column 2: variance (including the tip edge, excluding t2[i]).
#>   // p(y_i | y_-i) = N(r_i | mean_i, var_i + t2_i).
#>   matrix tree_loo_moments(vector r, vector w, real rv, vector t2,
#>                           array[] int anc, array[] int des,
#>                           int n_nodes, int root) {
#>     int N = num_elements(r);
#>     int E = num_elements(w);
#>     vector[n_nodes] um = rep_vector(0, n_nodes);
#>     vector[n_nodes] uv = rep_vector(0, n_nodes);
#>     vector[n_nodes] P = rep_vector(0, n_nodes);
#>     vector[n_nodes] H = rep_vector(0, n_nodes);
#>     vector[n_nodes] om = rep_vector(0, n_nodes);
#>     vector[n_nodes] ov = rep_vector(0, n_nodes);
#>     matrix[N, 2] out;
#>     um[1:N] = r;
#>     uv[1:N] = t2;
#>     // upward: likelihood message from the subtree below each node
#>     for (e in 1:E) {
#>       int c = des[e];
#>       int p = anc[e];
#>       real vv;
#>       if (c > N) {
#>         uv[c] = 1 / P[c];
#>         um[c] = H[c] / P[c];
#>       }
#>       vv = uv[c] + w[e];
#>       P[p] += 1 / vv;
#>       H[p] += um[c] / vv;
#>     }
#>     // downward: message from everything outside the subtree (preorder)
#>     om[root] = 0;
#>     ov[root] = rv;
#>     for (k in 1:E) {
#>       int e = E - k + 1;
#>       int c = des[e];
#>       int p = anc[e];
#>       real vv = uv[c] + w[e];
#>       real m_ex;
#>       real v_ex;
#>       if (ov[p] <= 0) {          // parent value known exactly (fixed root)
#>         m_ex = om[p];
#>         v_ex = 0;
#>       } else {
#>         real Pp = P[p] - 1 / vv + 1 / ov[p];
#>         real Hp = H[p] - um[c] / vv + om[p] / ov[p];
#>         m_ex = Hp / Pp;
#>         v_ex = 1 / Pp;
#>       }
#>       om[c] = m_ex;
#>       ov[c] = v_ex + w[e];
#>     }
#>     out[, 1] = om[1:N];
#>     out[, 2] = ov[1:N];
#>     return out;
#>   }
#> 
#>   // Exact leave-one-out predictive log densities, dense Gaussian model
#>   // (Sundararajan & Keerthi 2001; Burkner, Gabry & Vehtari 2021)
#>   vector gaussian_loo_lpd(vector r, matrix K) {
#>     int N = num_elements(r);
#>     matrix[N, N] Kinv = inverse_spd(K);
#>     vector[N] g = Kinv * r;
#>     vector[N] out;
#>     for (i in 1:N) {
#>       real s2 = 1 / Kinv[i, i];
#>       out[i] = normal_lpdf(r[i] | r[i] - g[i] * s2, sqrt(s2));
#>     }
#>     return out;
#>   }
#> 
#>   // Mean and variance of r_i given r_{-i} for every i, dense Gaussian model
#>   // (the variance includes all diagonal terms of K)
#>   matrix gaussian_loo_moments(vector r, matrix K) {
#>     int N = num_elements(r);
#>     matrix[N, N] Kinv = inverse_spd(K);
#>     vector[N] g = Kinv * r;
#>     matrix[N, 2] out;
#>     for (i in 1:N) {
#>       real s2 = 1 / Kinv[i, i];
#>       out[i, 1] = r[i] - g[i] * s2;
#>       out[i, 2] = s2;
#>     }
#>     return out;
#>   }
#> }
#> data {
#>   int<lower=1> N;
#>   int<lower=1> P;
#>   matrix[N, P] X;
#>   vector[N] y;
#>   vector<lower=0>[N] se;                // known measurement error SDs
#>   // tree, edges in postorder (children before parents)
#>   int<lower=1> E;
#>   int<lower=N + 1> n_nodes;
#>   int<lower=N + 1> root;
#>   array[E] int anc;
#>   array[E] int des;
#>   array[E] int first_child;
#>   array[N] int tip_edge;
#>   vector<lower=0>[E] len;
#>   vector<lower=0>[E] hs;
#>   vector<lower=0>[E] he;
#>   vector<lower=0, upper=1>[E] is_tip;
#>   real<lower=0> Tmax;
#>   int<lower=0, upper=1> use_prior;      // 0: posterior mode = maximum likelihood
#>   real<lower=0> prior_beta_scale;
#>   // bm1: bm()
#>   real<lower=0> bm1_sigma_scale;
#>   // white1: white()
#>   real<lower=0> white1_sigma_scale;
#> }
#> parameters {
#>   vector[P] coefs;              // regression coefficients
#>   real<lower=0> bm1_sigma;
#>   real<lower=0> white1_sigma;
#> }
#> model {
#>   {
#>     vector[E] w = rep_vector(0, E);
#>     real rv = 0;
#>     vector[N] t2 = square(se);
#>     // bm()
#>     {
#>     w += square(bm1_sigma) * (len);
#>   }
#>     t2 += square(white1_sigma);
#>     vector[N] r = y - X * coefs;
#>     target += tree_pruning_lpdf(r | w, rv, t2, anc, des, first_child, n_nodes, root);
#>   }
#>   if (use_prior == 1) {
#>     coefs ~ normal(0, prior_beta_scale);
#>     bm1_sigma ~ normal(0, bm1_sigma_scale);
#>     white1_sigma ~ normal(0, white1_sigma_scale);
#>   }
#> }
#> generated quantities {
#>   vector[N] log_lik;
#>   {
#>     vector[E] w = rep_vector(0, E);
#>     real rv = 0;
#>     vector[N] t2 = square(se);
#>     // bm()
#>     {
#>     w += square(bm1_sigma) * (len);
#>   }
#>     t2 += square(white1_sigma);
#>     vector[N] r = y - X * coefs;
#>     matrix[N, 2] mo = tree_loo_moments(r, w, rv, t2, anc, des, n_nodes, root);
#>     for (i in 1:N) {
#>       int e = tip_edge[i];
#>       real ri = r[i];
#>       real vi = mo[i, 2] + t2[i];
#>       log_lik[i] = normal_lpdf(ri | mo[i, 1], sqrt(vi));
#>     }
#>   }
#> }
```
